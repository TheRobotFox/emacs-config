(defun my/isearch-other-window (backward)
  "Search forward in the next window, or BACKWARD in the previous one."
  (interactive "P")
  (unless (one-window-p)
    (save-excursion
      (let ((next (if backward -1 1)))
        (other-window next)
        (if backward (isearch-backward) (isearch-forward))
        (other-window (- next))))))
