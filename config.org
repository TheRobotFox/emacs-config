This is the source for =init.el= and =early-init.el=. After editing, run
=org-babel-tangle= (=C-c C-v t=), then restart Emacs to check startup.
The files in =lisp/= are handwritten helpers and are loaded by this configuration.
Run the focused regression checks with:
=emacs -Q --batch -l tests/config-tests.el -f ert-run-tests-batch-and-exit=.

Keep package settings, hooks, and bindings together. Put shared editing commands
in =lisp/= and document their purpose here. The final =COMMENT Experiments=
subtree is inactive. To enable an experiment, move it to its owning section
and remove any explicit =:tangle no= header.

Intentional settings belong here. =custom.el= holds Custom's local bookkeeping
(such as trusted themes), and is optional on a fresh checkout.

Set startup memory limits and load optional local Custom bookkeeping.

(defvar bootstrap-version)
(let ((bootstrap-file
       (expand-file-name
        "straight/repos/straight.el/bootstrap.el"
        (or (bound-and-true-p straight-base-dir)
            user-emacs-directory)))

        (url-retrieve-synchronously
         "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
         'silent 'inhibit-cookies)
      (goto-char (point-max))
      (eval-print-last-sexp)))
  (load bootstrap-file nil 'nomessage))

Display C++ reference entries using the external =cppman= and =col= commands.

(defun my/cppman (entry)
  "View C++ reference entry using cppman in Emacs, similar to man command.
If called interactively with no argument, uses the symbol at point as default."
  (interactive
   (list (let* ((default (thing-at-point 'symbol t))
                (prompt (if default
                            (format "C++ Reference entry (default %s): " default)
                          "C++ Reference entry: ")))
           (read-string prompt nil nil default))))
  (let ((buffer (get-buffer-create (format "*cppman %s*" entry))))
    (with-current-buffer buffer
      (setq buffer-read-only nil)
      (erase-buffer)
      (call-process-shell-command
       (format "cppman %s | col -b" (shell-quote-argument entry))
       nil (current-buffer))
      (if (zerop (buffer-size))
          (message "No results found for %s" entry)
        (special-mode))
      (setq buffer-read-only t)
      (beginning-of-buffer))
    (unless (zerop (buffer-size))
      (switch-to-buffer buffer))))

**** Source/header creation

Create a companion implementation file from a C/C++ header.
For C++ headers, carry the enclosing namespace into the generated file.
#+begin_src emacs-lisp
(defun my/create-src-file ()
  "Create a new implementation buffer for the current C or C++ header."
  (interactive)
  (unless buffer-file-name
    (user-error "This buffer does not visit a file"))
  (let* ((file buffer-file-name)
         (ext (file-name-extension file))
         (new-ext (pcase ext ("h" "c") ("hpp" "cpp")
                         (_ (user-error "Expected a .h or .hpp header"))))
         (target (file-name-with-extension file new-ext))
         namespaces)
    (when (or (file-exists-p target) (get-file-buffer target))
      (user-error "Implementation file or buffer already exists: %s" target))
    ;; Inspect the header before switching buffers.  Plain C needs no parser.
    (when (equal ext "hpp")
      (unless (treesit-parser-list)
        (user-error "C++ namespace detection requires a Tree-sitter parser"))
      (let ((node (treesit-node-at (point))))
        (while node
          (when (equal (treesit-node-type node) "namespace_definition")
            (when-let* ((name (treesit-node-child-by-field-name node "name")))
              (push (treesit-node-text name t) namespaces)))
          (setq node (treesit-node-parent node)))))
    (let* ((namespace (string-join namespaces "::"))
           (contents (concat "#include \"" (file-name-nondirectory file) "\"\n\n"
                             (unless (string-empty-p namespace)
                               (format "namespace %s {\n} // %s" namespace namespace)))))
      (find-file-other-window target)
      (atomic-change-group (insert contents)))))
#+end_src

  (cl-defmethod org-roam-node-doom-filetitle ((node org-roam-node))
    "Return the value of \"#+title:\" (if any) from file that NODE resides in.
      If there's no file-level title in the file, return empty string."
    (or (if (= (org-roam-node-level node) 0)
            (org-roam-node-title node)
          (org-roam-node-file-title node))

    "Return hierarchy for NODE, constructed of its file title, OLP and direct title.
        If some elements are missing, they will be stripped out."
    (let ((title     (org-roam-node-title node))
          (olp       (org-roam-node-olp   node))
          (level     (org-roam-node-level node))
          (filetitle (org-roam-node-doom-filetitle node))

                   separator title))

        (t (concat (propertize filetitle 'face '(shadow italic))
                   separator (propertize (string-join olp separator) 'face '(shadow italic))
                   separator title)))))
